// Helper function to format current date
const formatDate = (date: Date): string => {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");
  return `${year}/${month}/${day}`;
};

        const { dx, dy } = gestureState;

        // For horizontal movements (swipe to delete)

          // Set the swiping ID for delete gestures

            setSwipingId(item.id);

      },
      onPanResponderMove: (_, gestureState) => {
        const { dx, dy } = gestureState;

        // Handle horizontal swipes for deletion

          // Only allow left swipes (negative dx)

            // Animate the swipe with dampening effect

                setDeleteModalVisible(true);
              }

        // Reset swipe state if we were swiping

            toValue: 0,

            tension: 40,
            useNativeDriver: true,

            setSwipingId(null);

        // Reset states on termination

          Animated.spring(swipeAnim, {
            toValue: 0,
            friction: 5,
            tension: 40,
            useNativeDriver: true,
          }).start(() => {
            setSwipingId(null);
          });
        }
      },
    })

    Animated.spring(scaleAnim, {
      toValue: 1,
      friction: 5,
      tension: 40,
      useNativeDriver: true,
    }).start(() => {
      setLongPressedId(null);
    });

      clearTimeout(longPressTimeout.current);
      longPressTimeout.current = null;

    item,
    index,

    item: ActivityWithTimer;
    index: number;

      transform: [

          scale: longPressedId === item.id ? scaleAnim : 1,

      shadowOpacity: 0.2,
      shadowRadius: 4,
      elevation: 5,
      zIndex: 1,

    // Create a simple pan responder to detect left swipe
    const itemPanResponder = PanResponder.create({
      onStartShouldSetPanResponder: () => false,
      onMoveShouldSetPanResponder: (_, gestureState) => {
        // Only respond to significant horizontal movement for swipe when allowed
        if (
          canSwipe &&
          Math.abs(gestureState.dx) > 10 &&
          Math.abs(gestureState.dx) > Math.abs(gestureState.dy)

          return true;

        return false;

      onPanResponderMove: (_, gestureState) => {
        // If swiped far enough to the left, show the delete UI
        if (gestureState.dx < SWIPE_THRESHOLD && !isSwipedLeft && canSwipe) {

        } else if (gestureState.dx > SWIPE_THRESHOLD && isSwipedLeft) {
          setSwipedState(item.id, false);
        }

      onPanResponderRelease: (_, gestureState) => {
        // If swiped far enough and released, show delete card
        if (gestureState.dx < SWIPE_THRESHOLD && canSwipe) {
          // We'll keep the swiped state but not automatically show the delete modal

          setSwipedState(item.id, false); // Reset state on release

                styles.deleteCardContent,

                e.stopPropagation(); // Prevent propagation to the TouchableWithoutFeedback

                // Immediate feedback

                // Immediately show the delete modal (no delay)

              onPressOut={() => setDeleteCardPressed(null)}

              delayPressOut={0} // No delay for visual feedback
              pressRetentionOffset={{
                top: 20,
                left: 20,
                bottom: 20,
                right: 20,
              }} // Larger touch area

              <Animated.View
                style={[
                  {
                    transform: [{ scale: deleteButtonScale }],
                    opacity: 1, // Using constant opacity instead of animated
                  },

                onPress={(e) => {
                  e.stopPropagation(); // Prevent propagation
                  setSwipedState(item.id, false);
                }}
                hitSlop={{ top: 20, right: 20, bottom: 20, left: 20 }} // Increased touch area
                activeOpacity={0.5} // More responsive visual feedback
              >
                <Icon name="close" size={16} color="#fff" />
              </TouchableOpacity>
            </TouchableOpacity>
          ) : (
            <TouchableOpacity
              style={styles.cardTouchable}
              onPress={() => {
                // Show JSON modal
                setSelectedActivity(item);
                setJsonModalVisible(true);

                // Mock API request
                const requestObject = {
                  method: "GET",
                  endpoint: `/api/activities/${item.id}`,
                  headers: {
                    Authorization:

                  success: true,
                  activity: {
                    ...item,
                    sessions: [

                        duration: "00:45",

                        date: formatDate(new Date()), // today
                        duration: item.currentTimer || "00:00",
                      },

                  },
                };

                // Log request and response

                setApiResponses((prev) => [...prev, responseObject]);
              }}
              onLongPress={() => !isFuture && handleLongPress(item)}
              onPressIn={() => handlePressIn(item.id)}
              onPressOut={handlePressOut}
              delayLongPress={LONG_PRESS_DURATION}
              disabled={isFuture}
              activeOpacity={0.9}
            >
              <LinearGradient
                colors={getGradientColors(item)}
                start={{ x: 0, y: 0 }}
                end={{ x: 1, y: 0 }}
                style={[
                  styles.activityContent,
                  isFuture && styles.futureActivityContent,
                ]}
              >
                <View style={styles.cardLeft}>
                  <Text
                    style={[
                      styles.activityTitle,
                      isFuture && styles.futureActivityText,
                    ]}
                  >
                    {item.title}
                    {isFuture && " (Future)"}
                  </Text>
                  <View style={styles.datesContainer}>
                    <View style={styles.dateContainer}>
                      <Icon
                        name="calendar-outline"
                        size={16}
                        color={isFuture ? "#aaa" : "#fff"}
                      />
                      <Text
                        style={[
                          styles.dateText,
                          isFuture && styles.futureActivityText,
                        ]}
                      >
                        {item.startDate}
                      </Text>
                    </View>
                    <View style={styles.dateContainer}>
                      <Icon
                        name="calendar-outline"
                        size={16}
                        color={isFuture ? "#aaa" : "#fff"}
                      />
                      <Text
                        style={[
                          styles.dateText,
                          isFuture && styles.futureActivityText,
                        ]}
                      >
                        {item.endDate}
                      </Text>
                    </View>
                  </View>
                </View>

                <View style={styles.cardRight}>
                  <View style={styles.categoryContainer}>
                    <Icon
                      name="person-outline"
                      size={14}
                      color={isFuture ? "#aaa" : "#fff"}
                    />
                    <Text
                      style={[
                        styles.categoryText,
                        isFuture && styles.futureActivityText,
                      ]}
                    >
                      {item.category}
                    </Text>
                  </View>

                  {!isFuture && (
                    <TouchableOpacity

                        styles.playButton,
                        item.isRunning && styles.playButtonActive,

                        e.stopPropagation(); // Prevent triggering the card's onPress

                      <Icon
                        name={item.isRunning ? "pause" : "play"}
                        size={20}
                        color="#fff"
                      />
                    </TouchableOpacity>
                  )}

                    <View style={styles.timerContainer}>
                      <Text style={styles.timerText}>{item.currentTimer}</Text>
                    </View>
                  )}
                </View>
              </LinearGradient>
            </TouchableOpacity>
          )}
        </View>
      </Animated.View>

  const handleAddActivity = (newActivity: Partial<Activity>) => {

      id: Date.now().toString(),

      currentTimer: "00:00",
      elapsedSeconds: 0,
      duration: "00:00",
      priority: activities.length + 1, // Add default priority as last item

      method: "POST",
      endpoint: "/api/activities",
      headers: {
        Authorization: "Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",

    setApiResponses((prev) => [...prev, responseObject]);

  // Function to handle the swipe gesture for an activity
  const handleSwipe = (
    id: string,
    gestureState: PanResponderGestureState,
    finish: () => void
  ) => {
    const { dx } = gestureState;

    // Create animation value if it doesn't exist
    if (!swipeAnimMap.has(id)) {
      swipeAnimMap.set(id, new Animated.Value(0));
    }

    const swipeAnim = swipeAnimMap.get(id)!;

    // Only handle left swipes (negative dx)
    if (dx < 0) {
      // Set swiping ID
      if (swipingId !== id) {
        setSwipingId(id);
      }

      // Update swipe animation - limit the swipe distance to create resistance
      const limitedDx = Math.max(dx, SWIPE_THRESHOLD * 1.2);
      swipeAnim.setValue(limitedDx);

      // If swiped far enough, show delete button
      if (dx <= SWIPE_THRESHOLD && showDeleteButton !== id) {
        setShowDeleteButton(id);
        // Also update our swipe state
        setSwipedState(id, true);
        // Provide strong haptic feedback
        Vibration.vibrate(40);
      } else if (dx > SWIPE_THRESHOLD && showDeleteButton === id) {
        setShowDeleteButton(null);
        setSwipedState(id, false);
      }

      // If swiped far enough to confirm delete
      if (dx <= CONFIRM_DELETE_THRESHOLD) {
        // Find activity
        const activity = activities.find((a) => a.id === id);
        if (activity && !activityToDelete) {
          // Vibrate to indicate delete action
          Vibration.vibrate(70);
          setActivityToDelete(activity);
          setDeleteModalVisible(true);

          // Reset the swipe
          Animated.spring(swipeAnim, {
            toValue: 0,
            friction: 5,
            tension: 40,
            useNativeDriver: true,
          }).start(() => {
            setSwipingId(null);
            setShowDeleteButton(null);
            setSwipedState(id, false);
            finish();
          });
        }
      }
    } else if (dx > 5) {
      // Reset on positive (right) swipe - only if we've moved noticeably to the right
      Animated.spring(swipeAnim, {
        toValue: 0,
        friction: 5,
        tension: 40,
        useNativeDriver: true,
      }).start(() => {
        setSwipingId(null);
        setShowDeleteButton(null);
        setSwipedState(id, false);
        finish();
      });
    }
  };

  // Function to reset swipe when released
  const resetSwipe = (id: string) => {
    if (swipeAnimMap.has(id)) {
      const swipeAnim = swipeAnimMap.get(id)!;
      Animated.spring(swipeAnim, {
        toValue: 0,
        friction: 5,
        tension: 40,
        useNativeDriver: true,
      }).start(() => {
        if (swipingId === id) {
          setSwipingId(null);
        }
        if (showDeleteButton === id && !deleteModalVisible) {
          setShowDeleteButton(null);
        }
        // Reset swiped state for this item
        setSwipedState(id, false);
      });
    }
  };

  // Function to dismiss all swiped items
  const dismissAllSwipedItems = () => {
    // Get all swiped item IDs
    const swipedIds = Object.keys(swipedItemIds).filter(
      (id) => swipedItemIds[id]
    );

    // Reset each one
    swipedIds.forEach((id) => {
      setSwipedState(id, false);
    });
  };

  return (
    <View style={styles.container}>
      <StatusBar barStyle="light-content" />
      <View style={styles.header}>
        <View style={styles.profileContainer}>
          <TouchableOpacity style={styles.profileImage}>
            <Icon name="person" size={24} color="#fff" />
          </TouchableOpacity>
          <Text style={styles.headerTitle}>Activity</Text>
        </View>
        <View style={styles.headerButtons}>
          <TouchableOpacity
            style={styles.jsonButton}
            onPress={() => setJsonModalVisible(true)}
          >
            <Icon name="code" size={22} color="#00E5FF" />
          </TouchableOpacity>
          <TouchableOpacity>
            <Icon
              name="stats-chart"
              size={24}
              color="#fff"
              onPress={() =>
                navigation.navigate("Stats", { title: "Statistics" })
              }
            />
          </TouchableOpacity>
        </View>
      </View>

      <View style={styles.tabContainer}>
        <TouchableOpacity
          style={[styles.tab, activeTab === "ACTIVITY" && styles.activeTab]}
          onPress={() => setActiveTab("ACTIVITY")}
        >
          <Text
            style={[
              styles.tabText,
              activeTab === "ACTIVITY" && styles.activeTabText,
            ]}
          >
            ACTIVITY
          </Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.tab, activeTab === "TASK" && styles.activeTab]}
          onPress={() => setActiveTab("TASK")}
        >
          <Text
            style={[
              styles.tabText,
              activeTab === "TASK" && styles.activeTabText,
            ]}
          >
            TASK
          </Text>
        </TouchableOpacity>
      </View>

      <TouchableOpacity
        style={styles.addButton}
        onPress={() => setModalVisible(true)}
      >
        <Icon name="add" size={24} color="#fff" />
        <Text style={styles.addButtonText}>Add new activity</Text>
      </TouchableOpacity>

      <View style={styles.instructionContainer}>
        <View style={styles.instructionItem}>
          <Icon name="arrow-back" size={16} color="#fff" />
          <Text style={styles.instructionText}>Swipe left to delete</Text>
        </View>
        <View style={styles.instructionItem}>
          <Icon name="finger-print" size={16} color="#fff" />
          <Text style={styles.instructionText}>Hold to delete</Text>
        </View>
        {Object.values(swipedItemIds).some((swiped) => swiped) && (
          <TouchableOpacity
            style={styles.dismissSwipesButton}
            onPress={dismissAllSwipedItems}
            activeOpacity={0.7}
          >
            <Icon name="close-circle" size={16} color="#fff" />
            <Text style={styles.instructionText}>Dismiss</Text>
          </TouchableOpacity>
        )}
      </View>

      {/* Activity List with tap anywhere to dismiss swiped cards */}
      <View style={{ flex: 1 }}>
        <FlatList
          data={activities}
          renderItem={renderActivityItem}
          keyExtractor={(item) => item.id}
          contentContainerStyle={styles.activityList}
          showsVerticalScrollIndicator={true}
          scrollEventThrottle={16}
          keyboardShouldPersistTaps="handled"
          CellRendererComponent={({ children, style, ...props }: any) => (
            <View style={style} {...props}>
              {children}
            </View>
          )}
          removeClippedSubviews={false} // Ensures animations aren't clipped
        />
      </View>

      {/* Using the reusable AddActivityModal component */}
      <AddActivityModal
        visible={modalVisible}
        onClose={() => setModalVisible(false)}
        onAdd={handleAddActivity}
      />

      {/* JSON Data Modal */}
      <Modal
        visible={jsonModalVisible}
        transparent={true}
        animationType="slide"
        onRequestClose={() => setJsonModalVisible(false)}
      >
        <View style={styles.jsonModalOverlay}>
          <View style={styles.jsonModalContent}>
            <View style={styles.jsonModalHeader}>
              <Text style={styles.jsonModalTitle}>API Data Overview</Text>
              <TouchableOpacity onPress={() => setJsonModalVisible(false)}>
                <Icon name="close" size={24} color="#fff" />
              </TouchableOpacity>
            </View>

            <ScrollView style={styles.jsonScrollView}>
              {selectedActivity ? (
                <View style={styles.selectedActivityInfo}>
                  <Text style={styles.jsonSectionTitle}>
                    Selected Activity:
                  </Text>
                  <View style={styles.jsonBox}>
                    <Text style={styles.jsonText}>
                      {JSON.stringify(selectedActivity, null, 2)}
                    </Text>
                  </View>
                </View>
              ) : null}

              <Text style={styles.jsonSectionTitle}>API Requests:</Text>
              {apiRequests.map((request, index) => (
                <View key={`req-${index}`} style={styles.jsonBox}>
                  <Text style={styles.jsonRequestTitle}>
                    {request.method} {request.endpoint}
                  </Text>
                  <Text style={styles.jsonText}>
                    {JSON.stringify(request, null, 2)}
                  </Text>
                </View>
              ))}

              <Text style={styles.jsonSectionTitle}>API Responses:</Text>
              {apiResponses.map((response, index) => (
                <View key={`res-${index}`} style={styles.jsonBox}>
                  <Text style={styles.jsonResponseTitle}>
                    Response {index + 1}
                  </Text>
                  <Text style={styles.jsonText}>
                    {JSON.stringify(response, null, 2)}
                  </Text>
                </View>
              ))}
            </ScrollView>
          </View>
        </View>
      </Modal>

      {/* Delete Confirmation Modal */}
      <Modal
        visible={deleteModalVisible}
        transparent={true}
        animationType="fade"
        onRequestClose={() => {
          setDeleteModalVisible(false);
          if (activityToDelete) {
            setSwipedState(activityToDelete.id, false);
          }
        }}
        statusBarTranslucent={true}
      >
        <TouchableWithoutFeedback
          onPress={() => {
            // Close modal when tapping outside
            setDeleteModalVisible(false);
            if (activityToDelete) {
              setSwipedState(activityToDelete.id, false);
            }
          }}
        >
          <View style={styles.deleteModalOverlay}>
            <TouchableWithoutFeedback onPress={(e) => e.stopPropagation()}>
              <View style={styles.deleteModalContent}>
                <View style={styles.deleteModalIconContainer}>
                  <Icon name="alert-circle" size={50} color="#FF3B30" />
                </View>
                <Text style={styles.deleteModalTitle}>Delete Activity</Text>

                <Text style={styles.deleteModalText}>
                  Are you sure you want to delete "{activityToDelete?.title}"?
                  This action cannot be undone.
                </Text>

                <View style={styles.deleteModalButtons}>
                  <TouchableOpacity
                    style={styles.deleteModalCancelButton}
                    onPress={() => {
                      setDeleteModalVisible(false);
                      if (activityToDelete) {
                        setSwipedState(activityToDelete.id, false);
                      }
                    }}
                    activeOpacity={0.7}
                    delayPressIn={0}
                  >
                    <Text style={styles.deleteModalCancelText}>Cancel</Text>
                  </TouchableOpacity>

                  <TouchableOpacity
                    style={styles.deleteModalDeleteButton}
                    onPress={handleDeleteActivity}
                    activeOpacity={0.7}
                    delayPressIn={0}
                  >
                    <Text style={styles.deleteModalDeleteText}>Delete</Text>
                  </TouchableOpacity>
                </View>
              </View>
            </TouchableWithoutFeedback>
          </View>
        </TouchableWithoutFeedback>
      </Modal>
    </View>
  );
};
